local shownErrors = {}
local function guard(fn)
    return function(...)
        local ok, err = pcall(fn, ...)
        if not ok then
            local key = tostring(err)
            if not shownErrors[key] then
                shownErrors[key] = true
                pcall(notify, "Erro: " .. key:sub(0, 80), ACCENT)
            end
        end
    end
end

local function safeNotify(...)
    pcall(notify, ...)
end

local function draggable(handle, target, onTap)
    local dragging, moved, startInput, startPos = false, false, nil, nil
    handle.InputBegan:Connect(guard(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging, moved = true, false
            startInput = input.Position
            startPos = target.Position
        end
    end))
    track(UIS.InputChanged:Connect(guard(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - startInput
            if d.Magnitude > 7 then moved = true end
            if moved then
                target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                    startPos.Y.Scale, startPos.Y.Offset + d.Y)
            end
        end
    end)))
    track(UIS.InputEnded:Connect(guard(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if dragging and not moved and onTap then onTap() end
        dragging = false
    end)))
end

local flyBV, flyBG
local upHeld, downHeld = false, false
local noclipSaved = {}

local function stopFly()
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
    local _, hum = getChar()
    if hum then hum.PlatformStand = false end
end

local function startFly()
    local _, hum, hrp = getChar()
    if not hum or not hrp then return end
    stopFly()
    if State.Stealth then return end
    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = hrp
    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    flyBG.P = 9e4
    flyBG.CFrame = hrp.CFrame
    flyBG.Parent = hrp
    hum.PlatformStand = true
end

local function restoreNoclip()
    for part, original in pairs(noclipSaved) do
        if part and part.Parent then part.CanCollide = original end
    end
    noclipSaved = {}
end

local function dirFromInput()
    local _, hum, hrp = getChar()
    if not hum or not hrp then return Vector3.zero end
    local cam = Workspace.CurrentCamera
    local dir = Vector3.zero
    local move = hum.MoveDirection
    if move.Magnitude > 0.05 and cam then
        local localMove = cam.CFrame:VectorToObjectSpace(move)
        dir = cam.CFrame.LookVector * (-localMove.Z) + cam.CFrame.RightVector * localMove.X
    end
    if upHeld then dir += Vector3.yAxis end
    if downHeld then dir -= Vector3.yAxis end
    if dir.Magnitude > 0.05 then dir = dir.Unit end
    return dir
end

local lastShot, aimTarget = 0, nil

local function rayVisible(part)
    local cam = Workspace.CurrentCamera
    if not cam then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LP.Character, cam }
    local hit = Workspace:Raycast(cam.CFrame.Position, part.Position - cam.CFrame.Position, params)
    if not hit then return true end
    return hit.Instance:FindFirstAncestorOfClass("Model") == part.Parent
end

local function findTarget()
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    local best, bestDist = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            local char = plr.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if char and hum and hum.Health > 0 then
                if (not State.TeamCheck) or plr.Team ~= LP.Team then
                    local head = char:FindFirstChild("Head")
                    local part = (State.AimPart == "Head" and head)
                        or char:FindFirstChild("HumanoidRootPart") or head
                    if part then
                        local sp, onScreen = cam:WorldToViewportPoint(part.Position)
                        if onScreen and sp.Z > 0 then
                            local dist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                            if dist <= State.AimFOV and dist < bestDist then
                                if (not State.WallCheck) or rayVisible(part) then
                                    best, bestDist = part, dist
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return best
end

track(RunService.RenderStepped:Connect(guard(function(dt)
    local t = os.clock()
    rootStroke.Transparency = 0.05 + math.sin(t * 2) * 0.12
    rootGlow.Transparency = 0.86 + math.sin(t * 2) * 0.08
    openerStroke.Transparency = math.abs(math.sin(t * 2.5)) * 0.15
    openerGlow.Transparency = 0.7 + math.sin(t * 2.5) * 0.12

    local _, hum, hrp = getChar()
    if not hum or not hrp then return end

    if State.Fly then
        local dir = dirFromInput()
        if State.Stealth then
            local vel = dir * State.FlySpeed
            vel += Vector3.new(0, Workspace.Gravity * dt, 0)
            hrp.AssemblyLinearVelocity = vel
            if hum:GetState() ~= Enum.HumanoidStateType.Freefall then
                hum:ChangeState(Enum.HumanoidStateType.Freefall)
            end
        else
            if not flyBV or not flyBV.Parent then startFly() end
            local cam = Workspace.CurrentCamera
            if flyBV and flyBG and cam then
                flyBV.Velocity = dir * State.FlySpeed
                flyBG.CFrame = CFrame.new(hrp.Position, hrp.Position + cam.CFrame.LookVector)
            end
        end
    end

    local cam = Workspace.CurrentCamera
    if State.AimOn and cam then
        local part = findTarget()
        aimTarget = part
        if part then
            local targetCF = CFrame.new(cam.CFrame.Position, part.Position)
            local alpha = 1 - math.clamp(State.AimSmooth / 100, 0, 0.92)
            cam.CFrame = cam.CFrame:Lerp(targetCF, math.clamp(alpha * dt * 45, 0.03, 1))
            if State.AutoFire and os.clock() - lastShot > 0.11 then
                lastShot = os.clock()
                local sp = cam:WorldToViewportPoint(part.Position)
                pcall(function() VirtualUser:ClickButton1(Vector2.new(sp.X, sp.Y)) end)
            end
        end
    else
        aimTarget = nil
    end
end)))

-- velocidade / pulo / noclip
local noclipClock = 0
track(RunService.Stepped:Connect(guard(function(dt)
    local char, hum, hrp = getChar()
    if not hum or not hrp then return end

    if State.SpeedOn then
        if State.Stealth then
            if hum.WalkSpeed ~= 16 then hum.WalkSpeed = 16 end
            if not State.Fly then
                local v = hrp.AssemblyLinearVelocity
                local horiz = Vector3.new(v.X, 0, v.Z)
                if horiz.Magnitude > 1 then
                    local target = horiz.Unit * State.Speed
                    hrp.AssemblyLinearVelocity = Vector3.new(target.X, v.Y, target.Z)
                end
            end
        else
            hum.WalkSpeed = State.Speed
        end
    end

    if State.JumpOn and not State.Stealth then
        hum.UseJumpPower = true
        hum.JumpPower = State.Jump
    end

    if State.Noclip then
        noclipClock += dt
        if noclipClock >= 0.15 then
            noclipClock = 0
            for _, part in ipairs(char:GetDescendants()) do

local vehCache, vehTime = {}, 0

local function collectVehicles(force)
    if not force and os.clock() - vehTime < 3 then return vehCache end
    local out = {}
    local rootFolder
    for _, name in ipairs({ "Vehicles", "Cars", "vehicles", "cars", "Vehicle" }) do
        local f = Workspace:FindFirstChild(name)
        if f then rootFolder = f break end
    end
    for _, d in ipairs((rootFolder or Workspace):GetDescendants()) do
        if d:IsA("Model") and d.PrimaryPart and d:FindFirstChildWhichIsA("VehicleSeat", true) then
            out[#out + 1] = d
        end
    end
    vehCache, vehTime = out, os.clock()
    return out
end

local function nearestVehicles(limit)
    local _, _, hrp = getChar()
    if not hrp then return {} end
    local origin = hrp.Position
    local scored = {}
    for _, model in ipairs(collectVehicles(false)) do
        local ok, pos = pcall(function() return model:GetPivot().Position end)
        if ok and pos then
            scored[#scored + 1] = { model = model, dist = (pos - origin).Magnitude }
        end
    end
    table.sort(scored, function(a, b) return a.dist < b.dist end)
    local result = {}
    for i = 1, math.min(limit, #scored) do result[i] = scored[i].model end
    return result
end

local function pullNearestVehicle()
    local _, _, hrp = getChar()
    if not hrp then return end
    local model = nearestVehicles(1)[1]
    if not model then return safeNotify("Nenhum veículo encontrado", ACCENT) end
    local ok = pcall(function()
        if model.PrimaryPart then
            pcall(function() model.PrimaryPart:SetNetworkOwner(LP) end)
        end
        model:PivotTo(hrp.CFrame * CFrame.new(0, 2, -12))
    end)
    safeNotify(ok and "Veículo puxado" or "Falha ao puxar", ok and GREEN or ACCENT)
end

local function pullManyVehicles()
    local _, _, hrp = getChar()
    if not hrp then return end
    local list = nearestVehicles(10)
    if #list == 0 then return safeNotify("Nenhum veículo encontrado", ACCENT) end
    for i, model in ipairs(list) do
        pcall(function()
            model:PivotTo(hrp.CFrame * CFrame.new(((i % 3) - 1) * 10, 3, -12 - math.floor(i / 3) * 12))
        end)
        task.wait(0.03)
    end
    safeNotify(#list .. " veículos puxados", GREEN)
end

local function enterNearestVehicle()
    local _, _, hrp = getChar()
    if not hrp then return end
    local model = nearestVehicles(1)[1]
    if not model then return safeNotify("Nenhum veículo encontrado", ACCENT) end
    local seat = model:FindFirstChildWhichIsA("VehicleSeat", true)
    if not seat then return safeNotify("Sem assento", ACCENT) end
    pcall(function() hrp.CFrame = seat.CFrame * CFrame.new(0, 3, 0) end)
    safeNotify("Teleportado ao veículo", GREEN)
end

local function isPlayerTool(tool)
    local anc = tool.Parent
    while anc do
        if anc:IsA("Model") and anc:FindFirstChildOfClass("Humanoid") then return true end
        anc = anc.Parent
    end
    return false
end

local function grabToolsFrom(container, onlyNearby)
    local _, _, hrp = getChar()
    if not hrp then return end
    local backpack = LP:FindFirstChildOfClass("Backpack")
    if not backpack then return safeNotify("Backpack não encontrada", ACCENT) end
    local count = 0
    for _, d in ipairs(container:GetDescendants()) do
        if d:IsA("Tool") and not isPlayerTool(d) then
            local ok = true
            if onlyNearby then
                local handle = d:FindFirstChild("Handle")
                ok = d:IsDescendantOf(Workspace) and handle
                    and (handle.Position - hrp.Position).Magnitude <= 60
            end
            if ok then
                count += 1
                local clone = d:Clone()
                clone.Name = d.Name .. " [" .. count .. "]"
                clone.Parent = backpack
                if count >= 40 then break end
                task.wait(0.02)
            end
        end
    end
    safeNotify(count > 0 and (count .. " armas adicionadas") or "Nenhuma arma encontrada",
        count > 0 and GREEN or ACCENT)
end

    sizeLimit.MaxTextSize = 11
    sizeLimit.MinTextSize = 7
    sizeLimit.Parent = b

    b.Activated:Connect(guard(function() selectTab(i) end))
    tabButtons[i] = b
end

selectTab(1)

-- ATALHOS
-- ======================================================================
track(UIS.InputBegan:Connect(guard(function(input, processed)
    if processed then return end
    local k = input.KeyCode
    if k == Enum.KeyCode.Space then upHeld = true end
    if k == Enum.KeyCode.E and State.Fly then upHeld = true end
    if k == Enum.KeyCode.LeftControl or k == Enum.KeyCode.Q then downHeld = true end
    if k == Enum.KeyCode.RightShift then togglePanel() end
    if k == Enum.KeyCode.C then
        State.AimOn = not State.AimOn
        if aimSetOn then aimSetOn(State.AimOn, false) end
    end
end)))
track(UIS.InputEnded:Connect(guard(function(input)
    local k = input.KeyCode
    if k == Enum.KeyCode.Space or k == Enum.KeyCode.E then upHeld = false end
    if k == Enum.KeyCode.LeftControl or k == Enum.KeyCode.Q then downHeld = false end
end)))

-- círculo do FOV
local fovCircle = Instance.new("Frame")
fovCircle.AnchorPoint = Vector2.new(0.5, 0.5)
fovCircle.Position = UDim2.new(0.5, 0, 0.5, 0)
fovCircle.BackgroundTransparency = 1
fovCircle.BorderSizePixel = 0
fovCircle.Visible = false
fovCircle.Parent = gui
round(fovCircle, 999999)
local fovStroke = stroke(fovCircle, ACCENT, 1.5, 0.35)

track(RunService.RenderStepped:Connect(guard(function()
    if State.AimOn then
        fovCircle.Visible = true
        fovCircle.Size = UDim2.new(0, State.AimFOV * 2, 0, State.AimFOV * 2)
        fovStroke.Color = aimTarget and ACCENT_HOT or ACCENT
        fovStroke.Transparency = aimTarget and 0.05 or 0.45
    else
        fovCircle.Visible = false
    end
end)))

local function cleanup()
    State.SpeedOn, State.JumpOn, State.Fly, State.Noclip = false, false, false, false
    State.InfJump, State.AimOn, State.ESP = false, false, false
    stopFly()
    restoreNoclip()
    clearESP()
    local _, hum = getChar()
    if hum then hum.WalkSpeed = 16 hum.JumpPower = 50 end
    for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    if gui then gui:Destroy() end
end
if getgenv then getgenv().ScarletCleanup = cleanup end
